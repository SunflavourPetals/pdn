cmake_minimum_required(VERSION 3.20)
project(pdn_headers_test LANGUAGES CXX)

# Minimal test project for the header-by-header review.
# Convention: one subdirectory per reviewed header, holding test.cpp plus a small
# add_executable/add_test pair. Tests include the real headers under ../../pdn/include,
# never a copy.

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(PDN_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../pdn/include")
set(PDN_INCLUDE_DIR "${PDN_INCLUDE_DIR}" CACHE PATH "PDN include root (parent of the directory holding pdn/spdn.h)")
if(NOT EXISTS "${PDN_INCLUDE_DIR}/pdn/spdn.h")
    message(FATAL_ERROR "PDN headers not found: ${PDN_INCLUDE_DIR}/pdn/spdn.h")
endif()

enable_testing()
